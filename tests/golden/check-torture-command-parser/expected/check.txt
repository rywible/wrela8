  Enum Opcode data
    variant Ping
    variant Get
    variant Set
    variant Delete
  Enum ParseError data deriving(From)
    variant BadOpcode(u8)
    fn from(source: take u8) -> ParseError
  Enum CommandError data deriving(From)
    variant Invalid(ParseError)
    fn from(source: take ParseError) -> CommandError

    field opcode: Opcode
    field length: u8
    init(mut self, code: read u8, length: read u8) -> Result[unit, ParseError]
    fn payload_len(read self) -> u8

    variant Get(key: u8)
    variant Set(key: u8, value: u8)
    variant Delete(key: u8)

  Fn describe(cmd: read Command) -> u8

    fn is_ok(read self) -> bool

  Fn unwrap_reply(slot: take Option[Reply], fallback: read u64) -> u64
  Fn first_matching[T](items: read [T; 3], target: read T) -> Option[usize]
  Fn use_first_matching_u8() -> Option[usize]
  Fn use_first_matching_bool() -> Option[usize]
  Fn count_gets(cmds: read [Command; 3]) -> u64

  Fn all_pings() -> u64
  Fn checked_reply(count: mut u64, raw: read [u8; 4]) -> u64
