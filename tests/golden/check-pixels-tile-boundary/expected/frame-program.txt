    # Exact rounded byte of `a*s + b*t + c >= 0` over the unit square.
    # Evaluating the final rational directly avoids the one-byte
    # enclosure introduced by an intermediate 512th-scale area.
    pub fn __wrela_pixels_p7_half_plane_byte(a: i32, b: i32, c: i32) -> [i64; 2]:
        if a < -1000000 or a > 1000000 or b < -1000000 or b > 1000000 or c < -1000000 or c > 1000000:
            return [0, 0]
        aa = a.to[i64]()
        bb = b.to[i64]()
        cc = c.to[i64]()
        if aa < 0:
            cc = cc + aa
            aa = -aa
        if bb < 0:
            cc = cc + bb
            bb = -bb
        threshold = -cc
        if threshold <= 0:
            return [1, 255]
        if threshold >= aa + bb:
            return [1, 0]
        numerator: i64 = 0
        denominator: i64 = 0
        if aa == 0:
            numerator = (bb - threshold) * 255
            denominator = bb
        elif bb == 0:
            numerator = (aa - threshold) * 255
            denominator = aa
        else:
            h0 = threshold * threshold
            h1: i64 = 0
            h2: i64 = 0
            if threshold > aa:
                h1 = (threshold - aa) * (threshold - aa)
            if threshold > bb:
                h2 = (threshold - bb) * (threshold - bb)
            denominator = 2 * aa * bb
            foreground = denominator - (h0 - h1 - h2)
            if foreground < 0:
                foreground = 0
            if foreground > denominator:
                foreground = denominator
            numerator = foreground * 255
        if not denominator > 0:
            return [0, 0]
        return [1, (2 * numerator + denominator) / (2 * denominator)]

    pub fn __wrela_pixels_p7_round_i32(value: f32) -> i32:
        if value < 0.0:
            return -((-value + 0.5).to[i32]())
        return (value + 0.5).to[i32]()

    pub fn __wrela_pixels_p7_min(a: f32, b: f32) -> f32:
        if a < b:
            return a
        return b

    pub fn __wrela_pixels_p7_max(a: f32, b: f32) -> f32:
        if a > b:
            return a
        return b

    pub fn __wrela_pixels_p7_clamp(value: f32, lo: f32, hi: f32) -> f32:
        return __wrela_pixels_p7_min(__wrela_pixels_p7_max(value, lo), hi)

    pub fn __wrela_pixels_p7_normalize_component(x: f32, y: f32, z: f32, component: usize) -> f32:
        length: f32 = sqrt_scalar(x * x + y * y + z * z)
        if not length > 0.0:
            return 0.0
        if component == 0:
            return x / length
        if component == 1:
            return y / length
        return z / length

    pub fn __wrela_pixels_p7_smooth_min(a: f32, b: f32, k: f32) -> f32:
        if not k > 0.0:
            return __wrela_pixels_p7_min(a, b)
        h: f32 = __wrela_pixels_p7_clamp(0.5 + 0.5 * (b - a) / k, 0.0, 1.0)
        return b + (a - b) * h - k * h * (1.0 - h)

    pub fn __wrela_pixels_p7_finite_or(value: f32, fallback: f32) -> f32:
        if value != value or value > 3.4028234663852886e38 or value < -3.4028234663852886e38:
            return fallback
        return value

    pub fn __wrela_pixels_p7_outward_low(value: f32) -> f32:
        bits = __wrela_pixels_f32_to_bits(value)
        if bits == 0:
            return __wrela_pixels_f32_from_bits(2147483649)
        if bits < 2147483648:
