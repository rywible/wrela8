    # Exact rounded byte of `a*s + b*t + c >= 0` over the unit square.
    # Evaluating the final rational directly avoids the one-byte
    # enclosure introduced by an intermediate 512th-scale area.
    pub fn __wrela_pixels_p7_half_plane_byte(a: i32, b: i32, c: i32) -> [i64; 2]:
        if a < -1000000 or a > 1000000 or b < -1000000 or b > 1000000 or c < -1000000 or c > 1000000:
            return [0, 0]
        aa = a.to[i64]()
        bb = b.to[i64]()
        cc = c.to[i64]()
        if aa < 0:
            cc = cc + aa
            aa = -aa
        if bb < 0:
            cc = cc + bb
            bb = -bb
        threshold = -cc
        if threshold <= 0:
            return [1, 255]
        if threshold >= aa + bb:
            return [1, 0]
        numerator: i64 = 0
        denominator: i64 = 0
        if aa == 0:
            numerator = (bb - threshold) * 255
            denominator = bb
        elif bb == 0:
            numerator = (aa - threshold) * 255
            denominator = aa
        else:
            h0 = threshold * threshold
            h1: i64 = 0
            h2: i64 = 0
            if threshold > aa:
                h1 = (threshold - aa) * (threshold - aa)
            if threshold > bb:
                h2 = (threshold - bb) * (threshold - bb)
            denominator = 2 * aa * bb
            foreground = denominator - (h0 - h1 - h2)
            if foreground < 0:
                foreground = 0
            if foreground > denominator:
                foreground = denominator
            numerator = foreground * 255
        if not denominator > 0:
            return [0, 0]
        return [1, (2 * numerator + denominator) / (2 * denominator)]

            # the raw grid's quantization noise (tens of raw units

            # geometric margins of subpixel features, which turned a
            # measure-zero validity boundary into a wide undecidable
            # skin. Soundness terms: the uv box radius, a mean-value
            # derivative bound over the q bracket, and a relative
            # rounding allowance far above the true f32 error.

                value_low = value_low * q_low + coefficient_value
                value_high = value_high * q_high + coefficient_value
                radius_eval = radius_eval * q_magnitude + coefficient_radius
                derivative_bound = derivative_bound * q_magnitude + magnitude
                magnitude = magnitude * q_magnitude + coefficient_magnitude + coefficient_radius
            spread = derivative_bound * (q_high - q_low)
            if spread < 0.0:
                return [0, 0]
            slack = radius_eval + spread + magnitude * 0.0000152587890625
            value_minimum = value_low
            if value_high < value_minimum:
                value_minimum = value_high
            value_maximum = value_low
            if value_high > value_maximum:
                value_maximum = value_high
            lower_bound = value_minimum - slack
            upper_bound = value_maximum + slack
            if not __wrela_pixels_p5_finite(lower_bound) or not __wrela_pixels_p5_finite(upper_bound):
                return [0, 0]
            definitely_valid = false
            definitely_invalid = false
            if sense[1] == 1:
                definitely_valid = upper_bound < 0.0
                definitely_invalid = lower_bound >= 0.0
            elif sense[1] == 2:
                definitely_valid = upper_bound <= 0.0
                definitely_invalid = lower_bound > 0.0
            elif sense[1] == 3:

                definitely_invalid = lower_bound > 0.000030517578125 or upper_bound < -0.000030517578125

                definitely_invalid = upper_bound < 0.0

                definitely_valid = lower_bound > 0.0
                definitely_invalid = upper_bound <= 0.0
            if definitely_invalid:
                return [1, 0]
            if not definitely_valid:
                ambiguous = true
            ordinal = ordinal + 1
        if ambiguous:
            return [2, 0]
        return [1, 1]
